local function safeOpenURL(url)
   local success, err = pcall(function()
       if setclipboard then
           setclipboard(url)
       end
       if syn and syn.request then
           syn.request({ Url = url, Method = "GET" })
       elseif request then
           request({ Url = url, Method = "GET" })
       end
       if pcall(function() game:GetService("MarketplaceService"):OpenBrowserWindow(url) end) then
           return
       end
   end)
   print("SYNC CHEATS: Link copiado ou aberto -> " .. url)
end

local function createESP(player)
   if ESPObjects[player] then return end
   local objects = { 
       Box = Drawing.new("Square"), 
       HealthOutline = Drawing.new("Square"), 
       HealthBar = Drawing.new("Square"), 
       Tracer = Drawing.new("Line"), 
       NameTag = Drawing.new("Text"),

   }
   objects.Box.Thickness = 1.2; objects.Box.Filled = false

local function removeESP(player)
   if ESPObjects[player] then
       local o = ESPObjects[player]
       pcall(function()
           o.Box:Remove(); o.HealthOutline:Remove(); o.HealthBar:Remove(); o.Tracer:Remove(); o.NameTag:Remove(); o.HeadDot:Remove()
           for _, l in ipairs(o.Skeleton) do l:Remove() end
       end)
       ESPObjects[player] = nil
   end
end

local raycastParams = RaycastParams.new()
raycastParams.FilterType = Enum.RaycastFilterType.Exclude
raycastParams.IgnoreWater = true

local function isVisible(targetPart, targetCharacter)
   if not getgenv().AimbotConfig.WallCheck then
       return true
   end
   
   local camPos = Camera.CFrame.Position
   local targetPos = targetPart.Position
   local ignoreList = {LocalPlayer.Character, targetCharacter}
   
   raycastParams.FilterDescendantsInstances = ignoreList
   local direction = targetPos - camPos
   local result = workspace:Raycast(camPos, direction, raycastParams)
   
   if result then
       local hitPart = result.Instance
       if hitPart:IsDescendantOf(targetCharacter) then
           return true
       end
       return false
   end
   
   return true
end

local function getClosestTarget()
   local target, shortest = nil, getgenv().AimbotConfig.FOVSize
   local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
   for _, p in ipairs(Players:GetPlayers()) do
       if p ~= LocalPlayer and p.Character then
           local hum = p.Character:FindFirstChildOfClass("Humanoid")
           if hum and hum.Health > 0 then
               local part = p.Character:FindFirstChild(getgenv().AimbotConfig.TargetPart) or p.Character:FindFirstChild("Head")
               if part then
                   local sPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                   if onScreen then
                       local dist = (Vector2.new(sPos.X, sPos.Y) - center).Magnitude
                       if dist < shortest then
                           if isVisible(part, p.Character) then
                               shortest, target = dist, part
                           end
                       end
                   end
               end
           end
       end
   end
   return target
end

RunService.RenderStepped:Connect(function()
   FOVCircle.Position = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
   FOVCircle.Visible = getgenv().AimbotConfig.ShowFOV

   local char = LocalPlayer.Character
   local hum = char and char:FindFirstChildOfClass("Humanoid")
   local hrp = char and char:FindFirstChild("HumanoidRootPart")

   if hum then
       if getgenv().MovementConfig.SpeedEnabled then hum.WalkSpeed = 24 end
       if getgenv().MovementConfig.BunnyHop and hum.FloorMaterial ~= Enum.Material.Air then
           hum:ChangeState(Enum.HumanoidStateType.Jumping)
       end
   end

   if getgenv().MovementConfig.SpinBot and hrp then
       hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(50), 0)
   end

   if getgenv().MovementConfig.NoClipEnabled and char then
       for _, p in ipairs(char:GetDescendants()) do if p:IsA("BasePart") then p.CanCollide = false end end
   end

   if getgenv().AimbotConfig.Enabled then
       local tPart = getClosestTarget()
       if tPart then
           if getgenv().AimbotConfig.AimLock then
               Camera.CFrame = CFrame.new(Camera.CFrame.Position, tPart.Position)
           else
               Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, tPart.Position), 1 / getgenv().AimbotConfig.Smoothness)
           end
       end
   end

   for player, objs in pairs(ESPObjects) do
       local pChar = player.Character
       local pHum = pChar and pChar:FindFirstChildOfClass("Humanoid")
       local pHrp = pChar and pChar:FindFirstChild("HumanoidRootPart")

       if pHum and pHum.Health > 0 and pHrp then
           local rootPos, onScreen = Camera:WorldToViewportPoint(pHrp.Position)
           if onScreen then
               local head = pChar:FindFirstChild("Head")
               local topPos = head and Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.8, 0)) or Camera:WorldToViewportPoint(pHrp.Position + Vector3.new(0, 3, 0))
               local bottomPos = Camera:WorldToViewportPoint(pHrp.Position - Vector3.new(0, 3.2, 0))
               
               local boxHeight = math.abs(topPos.Y - bottomPos.Y)
               local boxWidth = boxHeight / 2
               local boxPosition = Vector2.new(rootPos.X - boxWidth / 2, topPos.Y)

               if getgenv().ESPConfig.Box then
                   objs.Box.Size = Vector2.new(boxWidth, boxHeight)
                   objs.Box.Position = boxPosition
                   objs.Box.Color = Color3.fromRGB(255, 255, 255)
                   objs.Box.Visible = true
               else
                   objs.Box.Visible = false
               end

               if getgenv().ESPConfig.Health then
                   local healthPct = math.clamp(pHum.Health / pHum.MaxHealth, 0, 1)
                   local barHeight = boxHeight * healthPct
                   objs.HealthOutline.Size = Vector2.new(4, boxHeight + 2)
                   objs.HealthOutline.Position = Vector2.new(boxPosition.X - 7, boxPosition.Y - 1)
                   objs.HealthOutline.Visible = true

                   objs.HealthBar.Size = Vector2.new(2, barHeight)
                   objs.HealthBar.Position = Vector2.new(boxPosition.X - 6, boxPosition.Y + (boxHeight - barHeight))
                   objs.HealthBar.Color = Color3.fromRGB(255, 255, 255)
                   objs.HealthBar.Visible = true
               else
                   objs.HealthOutline.Visible = false
                   objs.HealthBar.Visible = false
               end

               if getgenv().ESPConfig.Tracers then
                   objs.Tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                   objs.Tracer.To = Vector2.new(rootPos.X, bottomPos.Y)
                   objs.Tracer.Color = Color3.fromRGB(255, 255, 255)
                   objs.Tracer.Visible = true
               else
                   objs.Tracer.Visible = false
               end

               if getgenv().ESPConfig.HeadDot and head then
                   local headVp, headVis = Camera:WorldToViewportPoint(head.Position)
                   if headVis then
                       objs.HeadDot.Position = Vector2.new(headVp.X, headVp.Y)
                       objs.HeadDot.Color = Color3.fromRGB(255, 255, 255)
                       objs.HeadDot.Visible = true
                   else
                       objs.HeadDot.Visible = false
                   end
               else
                   objs.HeadDot.Visible = false
               end

               if getgenv().ESPConfig.NameTag then
                   local displayName = getgenv().MiscConfig.StreamerMode and "Protected" or player.Name
                   local hpText = getgenv().ESPConfig.HealthText and " (" .. math.floor(pHum.Health) .. "HP)" or ""
                   objs.NameTag.Text = displayName .. hpText
                   objs.NameTag.Position = Vector2.new(rootPos.X, boxPosition.Y - 16)
                   objs.NameTag.Color = Color3.fromRGB(255, 255, 255)
                   objs.NameTag.Visible = true
               else
                   objs.NameTag.Visible = false
               end

               if getgenv().ESPConfig.Skeleton then
                   local joints = (pHum.RigType == Enum.HumanoidRigType.R15) and R15Joints or R6Joints
                   local idx = 1
                   local charModel = player.Character
                   if charModel then
                       for _, joint in ipairs(joints) do
                           local p1 = charModel:FindFirstChild(joint[1])
                           local p2 = charModel:FindFirstChild(joint[2])
                           local skelLine = objs.Skeleton[idx]
                           if p1 and p2 and skelLine then
                               local sp1, vis1 = Camera:WorldToViewportPoint(p1.Position)
                               local sp2, vis2 = Camera:WorldToViewportPoint(p2.Position)
                               if vis1 and vis2 then
                                   skelLine.From = Vector2.new(sp1.X, sp1.Y)
                                   skelLine.To = Vector2.new(sp2.X, sp2.Y)
                                   skelLine.Color = Color3.fromRGB(255, 255, 255)
                                   skelLine.Visible = true
                               else
                                   skelLine.Visible = false
                               end
                               idx = idx + 1
                           end
                       end
                   end
                   for i = idx, #objs.Skeleton do objs.Skeleton[i].Visible = false end
               else
                   for _, line in ipairs(objs.Skeleton) do line.Visible = false end
               end
           else
               objs.Box.Visible = false
               objs.HealthOutline.Visible = false
               objs.HealthBar.Visible = false
               objs.Tracer.Visible = false
               objs.NameTag.Visible = false
               objs.HeadDot.Visible = false
               for _, line in ipairs(objs.Skeleton) do line.Visible = false end
           end
       else
           objs.Box.Visible = false
           objs.HealthOutline.Visible = false
           objs.HealthBar.Visible = false
           objs.Tracer.Visible = false
           objs.NameTag.Visible = false
           objs.HeadDot.Visible = false
           for _, line in ipairs(objs.Skeleton) do line.Visible = false end
       end
   end
end)

local function VerifyKey()
   local enteredKey = KeyBox.Text or ""
   enteredKey = string.match(enteredKey, "^%s*(.-)%s*$")
   
   if enteredKey == "SyncV1" then
       if keySnowConn then keySnowConn:Disconnect() end
       KeyGui:Destroy()
       MainFrame.Visible = true
       ToggleButton.Visible = true
   else
       KeyErrorLabel.Text = "Chave incorreta! Utilize: SyncV1"
       TweenService:Create(KeyBox, TweenInfo.new(0.1), {BackgroundColor3 = Color3.fromRGB(40, 15, 15)}):Play()
       task.delay(0.3, function()
           if KeyBox and KeyBox.Parent then
               TweenService:Create(KeyBox, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(12, 12, 16)}):Play()
           end
       end)
   end
end

KeyBtn.MouseButton1Click:Connect(VerifyKey)

   if enterPressed then
       VerifyKey()
   end
end)
